## 5. 可复核计算方法

独立复算仅使用 Python 标准库 `sqlite3/json/decimal/statistics`，没有调用生产 `market_indicators` 实现。账户比例用 Decimal；区间收益为末收盘/首收盘−1；最近 N 个交易日使用 N+1 个实际收盘；均线用最后 N 个收盘的算术均值；波动是逐日简单收益的样本标准差（分母 n−1），未年化。下面片段可在仓库根目录执行，输出有界，不打印材料正文或事件上下文：

```python
import json, sqlite3
from pathlib import Path
from decimal import Decimal as D
from statistics import stdev

folder = Path("data/evaluations/eighth-final-v3-20261029")
db = sqlite3.connect((folder / "evaluation.sqlite3").resolve().as_uri() + "?mode=ro", uri=True)
def closes(material_id):
    body = json.loads(db.execute("select body from materials where id=?", (material_id,)).fetchone()[0])
    data = json.loads(body["content"])
    return {r["time"]: D(r["close"]) for r in data["rows"]}

a = closes("9f6ef97ec8d141469b02a9fb1b524cfc")
b = closes("d21ddfba155c4dcd86f4aff639391e09")
dates = sorted(a)
for n in (1, 5, 20, 60):
    window = dates[-(n + 1):]
    print("return", n, window[0], window[-1], (a[window[-1]] / a[window[0]] - 1) * 100)
print("MA20", sum(a[d] for d in dates[-20:]) / D(20))
for start in ("2026-08-01", "2026-09-01", "2026-09-15", "2026-09-23"):
    window = [d for d in dates if start <= d <= "2026-10-29"]
    common = [d for d in window if d in b]
    r = [float(a[y] / a[x] - 1) for x, y in zip(window, window[1:])]
    asset = a[common[-1]] / a[common[0]] - 1
    benchmark = b[common[-1]] / b[common[0]] - 1
    print(start, window[0], len(window), "vol_pct", stdev(r) * 100,
          "relative_pp", (asset - benchmark) * 100)
for units, cash in ((10000, 10000), (20000, 10000), (0, 20000)):
    value = D(units) * a[dates[-1]]
    total = value + D(cash)
    print(units, value, total, value / total * 100, D(cash) / total * 100)
db.close()
```
